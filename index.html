using UnityEngine; using System.Collections.Generic; [RequireComponent(typeof(Rigidbody))] public class GrandMountainCompleteEngine : MonoBehaviour { [Header("Skier Movement & Carving")] public float baseDownhillSpeed = 16f; public float steeringSensitivity = 95f; public float snowPlowFriction = 6f; private float speedMultiplier = 1f; [Header("Gravity & Slope Physics")] public float rigidSlopeGravity = 28f; public float surfaceCheckDistance = 1.3f; public LayerMask groundGeometryLayer; [Header("Trick State Telemetry")] public float apexJumpVelocity = 9f; public float axialFlipVelocity = 190f; public float axialSpinVelocity = 230f; private int ongoingComboScore = 0; private float airborneRotationTracker = 0f; [Header("GMA2 Camera & VFX Pipeline")] public Transform isometricCameraHandle; public Vector3 perspectiveOffset = new Vector3(0f, 38f, -26f); public float cameraTrackingDamp = 0.15f; public Material proceduralSnowMaterial; public Transform skiContactLeft; public Transform skiContactRight; public ParticleSystem powderSprayLeft; public ParticleSystem powderSprayRight; [Header("Procedural Resort World Map")] public Terrain targetTerrain; public GameObject skiLiftTowerPrefab; public GameObject liftCabinPrefab; // Enclosed cabin or chairlift model public GameObject slalomGatePrefab; public GameObject pineTreePrefab; public int totalSlalomGatesCount = 15; public int denseForestDensity = 300; [Header("Ski Lift Network Configuration")] public Transform liftBaseStation; public Transform liftSummitStation; public int totalCabinCount = 8; public int totalTowerCount = 4; public float liftCableVelocity = 5.0f; // Internal Skier States private Rigidbody structuralRb; private bool playerIsGrounded; private Vector3 operationalSlopeNormal; private bool leftPoleActive; private bool rightPoleActive; private bool airborneStateActive; // Internal Resort Map Lists private List activeLiftCabins = new List(); void Start() { // 1. Rigidbody & Controller Initialization structuralRb = GetComponent(); structuralRb.constraints = RigidbodyConstraints.FreezeRotation; structuralRb.useGravity = false; ConfigureIsometricCameraOrientation(); // 2. Map Generation Sequence if (targetTerrain == null) targetTerrain = Terrain.activeTerrain; GenerateResortInfrastructure(); GenerateSlalomRun(); PopulateMountainFlora(); } void Update() { CaptureControlInputs(); ProcessAirborneTricks(); } void FixedUpdate() { EvaluateTerrainNormals(); ApplyCustomSurfaceGravity(); if (playerIsGrounded) { airborneStateActive = false; ExecuteGroundCarving(); } else { airborneStateActive = true; } SynchronizeCameraPosition(); ModulateProceduralVFX(); AnimateSkiLiftNetwork(); } // ========================================== // SKIER INPUT & PHYSICS CONTROLLER SYSTEMS // ========================================== private void CaptureControlInputs() { leftPoleActive = Input.GetKey(KeyCode.A) || Input.GetKey(KeyCode.LeftArrow) || Input.GetButton("Fire1"); rightPoleActive = Input.GetKey(KeyCode.D) || Input.GetKey(KeyCode.RightArrow) || Input.GetButton("Fire2"); if ((Input.GetKeyDown(KeyCode.A) && Input.GetKey(KeyCode.D)) || Input.GetKeyDown(KeyCode.Space)) { if (playerIsGrounded) { structuralRb.AddForce(operationalSlopeNormal * apexJumpVelocity, ForceMode.VelocityChange); ongoingComboScore = 0; airborneRotationTracker = 0f; } } } private void EvaluateTerrainNormals() { RaycastHit terrainHit; if (Physics.Raycast(transform.position, -transform.up, out terrainHit, surfaceCheckDistance, groundGeometryLayer)) { playerIsGrounded = true; operationalSlopeNormal = terrainHit.normal; Quaternion surfaceTargetRotation = Quaternion.FromToRotation(transform.up, operationalSlopeNormal) * transform.rotation; transform.rotation = Quaternion.Slerp(transform.rotation, surfaceTargetRotation, Time.fixedDeltaTime * 12f); } else { playerIsGrounded = false; operationalSlopeNormal = Vector3.up; } } private void ApplyCustomSurfaceGravity() { structuralRb.AddForce(Vector3.down * rigidSlopeGravity, ForceMode.Acceleration); } private void ExecuteGroundCarving() { Vector3 directionalDownhillSlope = Vector3.ProjectOnPlane(transform.forward, operationalSlopeNormal).normalized; if (leftPoleActive && rightPoleActive) { structuralRb.linearDrag = 0.4f; structuralRb.AddForce(directionalDownhillSlope * baseDownhillSpeed * speedMultiplier, ForceMode.Acceleration); } else if (leftPoleActive) { transform.Rotate(Vector3.up, -steeringSensitivity * Time.fixedDeltaTime, Space.Self); structuralRb.AddForce(directionalDownhillSlope * (baseDownhillSpeed * 0.75f), ForceMode.Acceleration); } else if (rightPoleActive) { transform.Rotate(Vector3.up, steeringSensitivity * Time.fixedDeltaTime, Space.Self); structuralRb.AddForce(directionalDownhillSlope * (baseDownhillSpeed * 0.75f), ForceMode.Acceleration); } else { structuralRb.linearDrag = snowPlowFriction; } } private void ProcessAirborneTricks() { if (!airborneStateActive) return; float startingRotation = transform.rotation.eulerAngles.y; if (Input.GetKey(KeyCode.W)) transform.Rotate(Vector3.right, axialFlipVelocity * Time.deltaTime, Space.Self); else if (Input.GetKey(KeyCode.S)) transform.Rotate(Vector3.right, -axialFlipVelocity * Time.deltaTime, Space.Self); if (leftPoleActive && !rightPoleActive) transform.Rotate(Vector3.up, -axialSpinVelocity * Time.deltaTime, Space.World); else if (rightPoleActive && !leftPoleActive) transform.Rotate(Vector3.up, axialSpinVelocity * Time.deltaTime, Space.World); float rotationDelta = Mathf.Abs(transform.rotation.eulerAngles.y - startingRotation); if (rotationDelta > 180f) rotationDelta = 360f - rotationDelta; airborneRotationTracker += rotationDelta; if (airborneRotationTracker >= 360f) { ongoingComboScore += 180; airborneRotationTracker -= 360f; } } // ========================================== // TILT-SHIFT VISUALS & PARTICLES PIPELINE // ========================================== private void ConfigureIsometricCameraOrientation() { if (isometricCameraHandle == null) return; isometricCameraHandle.rotation = Quaternion.Euler(56f, 0f, 0f); } private void SynchronizeCameraPosition() { if (isometricCameraHandle == null) return; Vector3 targetCameraPosition = transform.position + perspectiveOffset; isometricCameraHandle.position = Vector3.Lerp(isometricCameraHandle.position, targetCameraPosition, cameraTrackingDamp); } private void ModulateProceduralVFX() { if (proceduralSnowMaterial != null && skiContactLeft != null && skiContactRight != null) { proceduralSnowMaterial.SetVector("_LeftSkiPosition", skiContactLeft.position); proceduralSnowMaterial.SetVector("_RightSkiPosition", skiContactRight.position); } if (powderSprayLeft == null || powderSprayRight == null) return; float kinematicVelocity = structuralRb.linearVelocity.magnitude; if (kinematicVelocity > 1.5f && playerIsGrounded) { var leftEmissionRate = powderSprayLeft.emission; var rightEmissionRate = powderSprayRight.emission; leftEmissionRate.rateOverTime = kinematicVelocity * 5.5f; rightEmissionRate.rateOverTime = kinematicVelocity * 5.5f; if (!powderSprayLeft.isPlaying) powderSprayLeft.Play(); if (!powderSprayRight.isPlaying) powderSprayRight.Play(); } else { powderSprayLeft.Stop(); powderSprayRight.Stop(); } } // ========================================== // PROCEDURAL RESORT LAYOUT & ENVIRONMENT MANAGEMENT // ========================================== private void GenerateResortInfrastructure() { if (liftBaseStation == null || liftSummitStation == null) return; // 1. Build Structural Support Cable Towers Along Mountain Incline if (skiLiftTowerPrefab != null && totalTowerCount > 0) { for (int i = 1; i 0) { for (int i = 0; i < totalCabinCount; i++) { float cabinRatio = (float)i / totalCabinCount; Vector3 spawnPosition = Vector3.Lerp(liftBaseStation.position, liftSummitStation.position, cabinRatio); // Add vertical cable line spacing buffer spawnPosition.y += 10f; GameObject spawnedCabin = Instantiate(liftCabinPrefab, spawnPosition, Quaternion.identity, transform); activeLiftCabins.Add(spawnedCabin.transform); } } } private void AnimateSkiLiftNetwork() { if (liftBaseStation == null || liftSummitStation == null) return; // Moves structural lift units steadily upwards towards peaks to create open world atmosphere foreach (Transform cabin in activeLiftCabins) { Vector3 targetNode = liftSummitStation.position; targetNode.y = cabin.position.y; // Maintain steady overhead line alignment cabin.position = Vector3.MoveTowards(cabin.position, targetNode, liftCableVelocity * Time.fixedDeltaTime); if (Vector3.Distance(new Vector3(cabin.position.x, 0, cabin.position.z), new Vector3(liftSummitStation.position.x, 0, liftSummitStation.position.z)) < 1.0f) { Vector3 resettingBasePoint = liftBaseStation.position; resettingBasePoint.y = cabin.position.y; cabin.position = resettingBasePoint; } } } private void GenerateSlalomRun() { if (slalomGatePrefab == null || liftSummitStation == null) return; Vector3 gateSpawnCursor = liftSummitStation.position - new Vector3(8f, 0f, 8f); for (int i = 0; i < totalSlalomGatesCount; i++) { if (targetTerrain != null) { float sampledHeight = targetTerrain.SampleHeight(gateSpawnCursor); gateSpawnCursor.y = sampledHeight; } GameObject gateInstance = Instantiate(slalomGatePrefab, gateSpawnCursor, Quaternion.identity, transform); gateInstance.name = $"Resort_Slalom_Gate_{i}"; float lateralWeaveFactor = (i % 2 == 0) ? 14f : -14f; gateSpawnCursor += new Vector3(lateralWeaveFactor, 0f, -24f); } } private void PopulateMountainFlora() { if (pineTreePrefab == null || targetTerrain == null) return; TerrainData terrainData = targetTerrain.terrainData; for (int i = 0; i < denseForestDensity; i++) { float targetSampleX = Random.Range(0f, terrainData.size.x); float targetSampleZ = Random.Range(0f, terrainData.size.z); Vector3 potentialPos = new Vector3(targetSampleX, 0f, targetSampleZ) + targetTerrain.transform.position; potentialPos.y = targetTerrain.SampleHeight(potentialPos); bool spacingConflict = false; if (liftBaseStation != null && Vector3.Distance(potentialPos, liftBaseStation.position) < 18f) spacingConflict = true; if (liftSummitStation != null && Vector3.Distance(potentialPos, liftSummitStation.position) < 18f) spacingConflict = true; if (!spacingConflict) { GameObject treeNode = Instantiate(pineTreePrefab, potentialPos, Quaternion.Euler(0, Random.Range(0, 360), 0), transform); treeNode.transform.localScale = Vector3.one * Random.Range(0.8f, 1.4f); } } } }
